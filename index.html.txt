    <script>
        const canvas = document.getElementById('canvas-fractal');
        const ctx = canvas.getContext('2d');
        function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
        window.addEventListener('resize', resize);
        resize();
        let time = 0;
        function drawFibonacciSpiral() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.strokeStyle = 'rgba(0, 255, 255, 0.1)';
            ctx.lineWidth = 1.5;
            const centerX = canvas.width / 2;
            const centerY = canvas.height / 2;
            ctx.beginPath();
            let a = 0, b = 1, angle = 0;
            for (let i = 0; i < 18; i++) {
                const radius = b * (2 + Math.sin(time * 0.2)) * 4;
                const x = centerX + Math.cos(angle + time * 0.05) * radius;
                const y = centerY + Math.sin(angle + time * 0.05) * radius;
                if (i === 0) ctx.moveTo(x, y);
                else ctx.lineTo(x, y);
                let temp = a + b; a = b; b = temp;
                angle += Math.PI / 2;
            }
            ctx.stroke();
            time += 0.02;
            requestAnimationFrame(drawFibonacciSpiral);
        }
        drawFibonacciSpiral();
    </script>
